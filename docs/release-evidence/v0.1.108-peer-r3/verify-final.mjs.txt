import { readFileSync, readdirSync, writeFileSync, copyFileSync, mkdtempSync, mkdirSync } from 'node:fs'
import { join, resolve } from 'node:path'
import { tmpdir } from 'node:os'
import { createHash } from 'node:crypto'
import { spawnSync } from 'node:child_process'
import { _electron as electron, expect } from '@playwright/test'
import { auditPackagePrivacy, developmentPathStrings } from '../../tools/package-privacy-audit.mjs'
const evidence = resolve('test-results/peer-distribution-0.1.108')
const bundle = readFileSync(join(evidence, 'final-extracted-path.txt'), 'utf8')
const oldBundle = readFileSync(join(evidence, 'network-tested-path.txt'), 'utf8')
const hash = path => createHash('sha256').update(readFileSync(path)).digest('hex')
const walk = path => readdirSync(path, { withFileTypes: true }).flatMap(entry => entry.isDirectory() ? walk(join(path, entry.name)) : [join(path, entry.name)])
const files = walk(bundle)
auditPackagePrivacy(files, developmentPathStrings(process.cwd()))
writeFileSync(join(evidence, 'final-privacy-audit.json'), JSON.stringify({ status: 'passed', fileCount: files.length, scope: 'final ZIP extracted files; on repeated checks also includes the verified upstream downloads', archiveUserDataIncluded: false }, null, 2))
const script = join(bundle, '首次启动.ps1')
if (hash(script) !== hash(join(oldBundle, '首次启动.ps1'))) throw new Error('Network-tested first-run script differs from final ZIP')
const downloadHashes = {}
mkdirSync(join(bundle, '程序/resources/models'), { recursive: true })
for (const name of ['ffmpeg.exe', 'ffprobe.exe']) {
  const source = join(oldBundle, '程序/resources/models', name)
  copyFileSync(source, join(bundle, '程序/resources/models', name))
  downloadHashes[name] = hash(source)
}
const ps = join(process.env.SystemRoot, 'System32/WindowsPowerShell/v1.0/powershell.exe')
const repeat = spawnSync(ps, ['-NoProfile', '-ExecutionPolicy', 'Bypass', '-Command', '[Console]::OutputEncoding=New-Object Text.UTF8Encoding; [Net.WebRequest]::DefaultWebProxy=New-Object Net.WebProxy("http://127.0.0.1:1"); & $env:SHORTDRAMA_PEER_TEST_SCRIPT -PrepareOnly'], { encoding: 'utf8', windowsHide: true, timeout: 30000, env: { ...process.env, SHORTDRAMA_PEER_TEST_SCRIPT: script } })
if (repeat.status !== 0) {
  writeFileSync(join(evidence, 'repeat-first-run-failure.log'), (repeat.stdout ?? '') + (repeat.stderr ?? ''))
  throw new Error('Offline second-start check failed')
}
writeFileSync(join(evidence, 'repeat-first-run.log'), repeat.stdout + repeat.stderr)
const broken = mkdtempSync(join(tmpdir(), 'shortdrama-peer-bad-hash-'))
mkdirSync(join(broken, '程序/resources/models'), { recursive: true })
copyFileSync(script, join(broken, '首次启动.ps1'))
writeFileSync(join(broken, '程序/短剧投流素材生成平台.exe'), '')
writeFileSync(join(broken, '程序/resources/models/ffmpeg.exe'), 'deliberately corrupt fixture')
const rejected = spawnSync(ps, ['-NoProfile', '-ExecutionPolicy', 'Bypass', '-Command', '[Console]::OutputEncoding=New-Object Text.UTF8Encoding; & $env:SHORTDRAMA_PEER_TEST_SCRIPT -PrepareOnly'], { encoding: 'utf8', windowsHide: true, timeout: 30000, env: { ...process.env, SHORTDRAMA_PEER_TEST_SCRIPT: join(broken, '首次启动.ps1') } })
if (rejected.status !== 1 || !rejected.stdout.includes('校验不通过')) throw new Error('Corrupted component was not rejected')
writeFileSync(join(evidence, 'corrupt-component-rejected.log'), rejected.stdout + rejected.stderr)
const userData = mkdtempSync(join(tmpdir(), 'shortdrama-peer-app-smoke-'))
const executablePath = join(bundle, '程序/短剧投流素材生成平台.exe')
const app = await electron.launch({ executablePath, args: ['--packaged-smoke', '--packaged-smoke-user-data', userData], env: { ...process.env, PATH: `${process.env.SystemRoot}\\System32;${process.env.SystemRoot}`, SHORTDRAMA_MATERIAL_CHANNEL: 'packaged-smoke' } })
try {
  const page = await app.firstWindow()
  await expect(page.getByRole('heading', { name: '剧目库', exact: true })).toBeVisible()
  const data = await page.evaluate(async () => ({ health: await window.desktopApi.getAppHealth(), dramas: await window.desktopApi.listDramas(), providers: await window.desktopApi.listProviderProfiles() }))
  if (data.dramas.length !== 0 || data.providers.length !== 0 || !data.health.databaseInitialized || data.health.appVersion !== '0.1.108') throw new Error('Fresh account is not empty and healthy')
  await page.screenshot({ path: join(evidence, 'final-empty-library.png') })
  await page.getByRole('button', { name: '设置', exact: true }).click()
  const settings = page.frameLocator('iframe[title="设置"]')
  await expect(settings.getByRole('heading', { name: '能力供应商', exact: true })).toBeVisible()
  await expect(settings.getByRole('button', { name: '进入独立配置', exact: true })).toHaveCount(3)
  await page.screenshot({ path: join(evidence, 'final-provider-setup.png') })
  const packaged = await app.evaluate(({ app: runtime }) => runtime.isPackaged)
  if (!packaged) throw new Error('Not a packaged executable')
  writeFileSync(join(evidence, 'final-isolated-smoke.json'), JSON.stringify({ status: 'passed', appVersion: data.health.appVersion, isPackaged: packaged, databaseInitialized: true, dramaCount: 0, providerCount: 0, systemOnlyPath: true, setupEntryVisible: true, firstRunScriptHash: hash(script), networkDownloadedComponentHashes: downloadHashes, repeatWithoutNetwork: true, corruptComponentRejected: true, verificationScope: 'host isolated installation, not clean VM' }, null, 2))
} finally { await app.close() }
console.log('FINAL_PEER_ISOLATED_SMOKE_PASSED')
