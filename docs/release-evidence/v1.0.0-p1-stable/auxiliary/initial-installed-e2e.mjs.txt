import { readFileSync } from 'node:fs'
import { resolve, dirname, join } from 'node:path'
import { spawnSync } from 'node:child_process'
import { fileSha256 } from '../tools/candidate-provenance.mjs'
const installed = JSON.parse(readFileSync('.tmp/initial-installed.json', 'utf8'))
const manifest = JSON.parse(readFileSync('release/candidates/v1.0.0-p1-stable/candidate-manifest.json', 'utf8'))
if (installed.releaseId !== manifest.releaseId || fileSha256(installed.executable) !== manifest.executableSha256 || fileSha256(join(dirname(installed.executable), 'resources/app.asar')) !== manifest.appAsarSha256) throw Error('Installed identity mismatch')
console.log('ACTUAL_INSTALLED_TEST_IDENTITY=' + JSON.stringify(installed))
const result = spawnSync(process.execPath, ['node_modules/@playwright/test/cli.js', 'test', 'tests/current/packaged-real-dpi.spec.ts', 'tests/current/packaged-business-smoke.spec.ts', 'tests/current/original-clip-pool-export-report.spec.ts', 'tests/current/prototype-navigation-transition.spec.ts', '--workers=2'], { stdio: 'inherit', windowsHide: true, env: { ...process.env, SHORTDRAMA_PACKAGED_EXE: installed.executable, PLAYWRIGHT_OUTPUT_DIR: resolve('test-results/initial-release-1.0.0/candidate-installed') } })
if (result.error) throw result.error
process.exitCode = result.status ?? 1
